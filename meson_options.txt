# General options

option('enable_docs',
  type: 'boolean',
  value: false,
  description: 'Build (with Doxygen) and install API documentation')

option('enable_examples',
    type: 'boolean',
    value: false,
    description: 'Build the csstest and dvd_region diagnostic tools')

option('enable_tests',
    type: 'boolean',
    value: true,
    description: 'Build and register automated libdvdcss tests')
